Build and Run with Docker
bash
# Build the Docker image
docker build -t chiang-mai-business .

# Run the container
docker run -p 8080:8080 chiang-mai-business

# Or run with environment variables
docker run -p 8080:8080 -e GIN_MODE=release chiang-mai-business
🏃 Local Development
bash
# Build CSS with Tailwind (requires Node.js/npm)
npx tailwindcss -i ./static/css/tailwind.css -o ./static/css/styles.css --watch

# Or use the standalone CLI (if downloaded)
./tailwindcss -i ./static/css/tailwind.css -o ./static/css/styles.css --watch

# Run the Go application
go run cmd/server/main.go
